import streamlit as st
import pandas as pd
import folium
from streamlit_folium import st_folium

# Sayfa Yapılandırması
st.set_page_config(
    page_title="Gemi & Elektrik Enspeksiyon Takip Paneli",
    page_icon="🚢",
    layout="wide"
)

st.title("🚢 Deniz Elektrik Enspektörlüğü - Gemi & Denetim Takip Paneli")
st.caption("Gemi konumları, izolasyon (Megger) test durumları ve pano sağlık skorları")

# 1. Örnek Veri Seti (Gerçek uygulamada API veya veritabanından çekilebilir)

    return pd.DataFrame([
        {
            "mmsi": 271000123,
            "name": "MV PACIFIC STAR",
            "type": "Konteyner",
            "lat": 40.985,
            "lon": 28.780,
            "port": "Ambarlı Limanı",
            "status": "BAKIM GEREKİYOR",
            "megger_val": "0.3 MΩ",
            "last_inspection": "2026-08-10",

        },
        {

        },

            "lat": 40.850,
            "lon": 29.280,
            "port": "Yılport / Dilovası",
            "status": "KRİTİK / TAKİP",
            "megger_val": "0.8 MΩ",
            "last_inspection": "2026-09-01",
            "notes": "Acil durum panosu (Em'cy SWBD) izleme altında."
        }
    ])

df = get_vessel_data()

# 2. Sol Kenar Çubuğu (Filtreleme Alanı)
st.sidebar.header("🔍 Gemi Filtreleri")

# Durum Filtresi
status_list = ["Tümü"] + list(df["status"].unique())
selected_status = st.sidebar.selectbox("Enspeksiyon Durumu", status_list)

# Gemi Tipi Filtresi
type_list = ["Tümü"] + list(df["type"].unique())
selected_type = st.sidebar.selectbox("Gemi Tipi", type_list)

# Arama Kutusu
search_query = st.sidebar.text_input("Gemi Adı / MMSI Ara", "")

# Filtreleme Mantığı
filtered_df = df.copy()

if selected_status != "Tümü":
    filtered_df = filtered_df[filtered_df["status"] == selected_status]

if selected_type != "Tümü":
    filtered_df = filtered_df[filtered_df["type"] == selected_type]

if search_query:
    filtered_df = filtered_df[
        filtered_df["name"].str.contains(search_query, case=False) | 
        filtered_df["mmsi"].astype(str).contains(search_query)
    ]

# 3. Ana Panel Düzeni (Harita ve Veri Tablosu)
col1, col2 = st.columns([2, 1])

with col1:
    st.subheader("🗺️ Canlı Harita (Marmara Bölgesi)")
    
    # Harita Merkezini Ayarla
    m = folium.Map(location=[40.85, 29.0], zoom_start=9, tiles="CartoDB positron")

    # Duruma Göre Harita İkon Renkleri
    color_map = {
        "UYGUN": "green",
        "KRİTİK / TAKİP": "orange",
        "BAKIM GEREKİYOR": "red"
    }

    # Haritaya Gemileri Ekle
    for _, row in filtered_df.iterrows():
        popup_html = f"""
        <div style="font-family: Arial; width: 180px;">
            <h4>{row['name']}</h4>
            <b>Tip:</b> {row['type']}<br>
            <b>MMSI:</b> {row['mmsi']}<br>
            <b>Liman:</b> {row['port']}<br>
            <b>Megger:</b> {row['megger_val']}<br>
            <b>Durum:</b> <span style="color:{color_map.get(row['status'], 'blue')};"><b>{row['status']}</b></span><br>
            <b>Not:</b> {row['notes']}
        </div>
        """
        
        folium.Marker(
            location=[row["lat"], row["lon"]],
            popup=folium.Popup(popup_html, max_width=250),
            tooltip=f"{row['name']} - ({row['status']})",
            icon=folium.Icon(color=color_map.get(row['status'], 'blue'), icon='ship', prefix='fa')
        ).add_to(m)

    # Streamlit Üzerinde Haritayı Büyüklüğüyle Göster
    st_folium(m, width="100%", height=500)

with col2:
    st.subheader("📋 Seçili Gemi Detayları")
    st.metric(label="Toplam Takip Edilen Gemi", value=len(filtered_df))
    
    if not filtered_df.empty:
        for _, row in filtered_df.iterrows():
            with st.expander(f"🚢 {row['name']} ({row['status']})"):
                st.write(f"**Liman:** {row['port']}")
                st.write(f"**Son İzolasyon Değeri:** {row['megger_val']}")
                st.write(f"**Son Denetim:** {row['last_inspection']}")
                st.info(f"**Enspektör Notu:** {row['notes']}")
    else:
        st.warning("Filtrelere uygun gemi bulunamadı.")

# 4. Alt Tablo
st.subheader("📊 Tüm Filo Denetim Listesi")
st.dataframe(filtered_df[["name", "mmsi", "type", "port", "status", "megger_val", "last_inspection"]], use_container_width=True)
